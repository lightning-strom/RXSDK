版本：v3.5

本模块提供接入鸿蒙 Nex的游戏近场快传能力。 系统版本要求

  • 鸿蒙系统起始版本：5.1.0(API Level 18)
  • 兼容性校验：可通过 NearbyTransferService.isSupport() 方法检测当前设备是否满足版本要求
  • 更多功能说明 ：参照华为官方文档

导入模块

import { NearbyTransferService, NearbyEventType } from 'hmssdk';

开发准备

在工程的entry模块module.json5文件中，新增metadata并配置app_id，同时新增requestPermissions并配置如下权限。

"module": {
  "name": "entry",
  "type": "entry",
  "description": "xxxx",
  "mainElement": "xxxx",
  "deviceTypes": [
    "phone"
  ],
  "deliveryWithInstall": true,
  "pages": "$profile:main_pages",
  "abilities": [],
  "metadata": [ // 配置如下信息
    {
      "name": "app_id",
      "value": "xxxxxx" // 配置为前面步骤中获取的APP ID
    }
  ],
   "requestPermissions": [ // 配置权限
     {
       "name": "ohos.permission.INTERNET" // 允许使用Internet网络权限
     },
     {
       "name": "ohos.permission.GET_NETWORK_INFO"  // 允许应用获取数据网络信息权限
     },
     {
       "name": "ohos.permission.SET_NETWORK_INFO" // 允许应用配置数据网络权限
     },
     {
       "name": "ohos.permission.DISTRIBUTED_DATASYNC", // 允许不同设备间的数据交换权限
       "reason": "$string:distributed_permission",
       "usedScene": {
         "abilities": [
           "EntryAbility"
         ],
         "when": "inuse"
       }
     }
   ]
}

基础类型定义

参考手册 应用沙箱目录

/**
 * 附近传输服务事件类型枚举
 * 定义了所有可能触发的事件类型，覆盖初始化、连接、发现、传输、错误等场景
 */
enum NearbyEventType {
  /** 初始化事件：服务初始化完成/失败时触发 */
  init = "init",
  /** 连接通知事件：设备连接状态变化时触发（如连接成功、断开等） */
  connectNotify = "connectNotify",
  /** 发现事件：发送端执行附近设备发现操作时触发 */
  discovery = "discovery",
  /** 错误事件：服务运行过程中出现错误时触发 */
  error = "error",
  /** 传输通知事件：数据传输状态变化时触发（如开始传输、传输完成等） */
  transferNotify = "transferNotify",
  /** 传输数据包事件：接收到或发送数据包时触发 */
  transferPackageData = "transferPackageData",
  /** 接收包信息事件：收到其他设备发送的包信息时触发 */
  receivePackageInfo = "receivePackageInfo",
  /** 回复包信息结果事件：回复包信息后的结果回调 */
  replyPackageInfoResult = "replyPackageInfoResult",
  /** 发送包信息事件：发送包信息操作触发 */
  sendPackageInfo = "sendPackageInfo",
  /** 销毁事件：服务销毁时触发 */
  destroy = "destroy",
  /** 自动绑定附近游戏事件：触发自动绑定附近游戏逻辑时回调 */
  autoBindNearbyGame = "autoBindNearbyGame",
  /** 发布附近游戏事件：发布游戏到附近设备时触发 */
  publishNearbyGame = "publishNearbyGame",
  /** 绑定附近游戏事件：主动绑定附近游戏时触发 */
  bindNearbyGame = "bindNearbyGame",

}

/**
 * 事件回调参数接口
 * 所有 NearbyEventType 事件触发时传递的参数结构
 */
interface EventArgs {
  /** 事件类型，对应 NearbyEventType 枚举值 */
  event: NearbyEventType;

  /** 事件附带数据，可选，根据不同事件类型携带不同业务数据 */
  data?: ESObject;

  /** 状态码：0 表示成功，非 0 表示错误码 */
  code: number;

  /** 事件描述信息：错误时为错误提示，成功时可为空 */
  message?: string;
}

type NearbyEvent = (params: EventArgs) => void;

/**
 * 注册附近传输服务的事件回调函数

 *                回调参数为 EventArgs 类型，包含事件类型、状态码、数据及错误信息等
 */
public registerCallback(event?: NearbyEvent)

          } else {
            this.showText(`收到连接，当前为${p} `, result);
          }

          this.showText(`连接状态异常， code= ${result.code},message=` + result.message);
          // await NearbyTransferService.getInstance().destroy();
          // TransferDialogManager.getInstance(uiContext)?.destroy();
        }
      } else if (result.event == NearbyEventType.receivePackageInfo) {
        let pi = result.data as gameNearbyTransfer.PackageInfo
        this.showText(`收到包信息`, result.data);

        let packageInfoResult: gameNearbyTransfer.PackageInfoResult = {
          packageInfoResultCode: gameNearbyTransfer.PackageInfoResultCode.PACKAGE_AVAILABLE_COMPARED,
          message: "包信息比对结果"
        };
        NearbyTransferService.getInstance().replyPackageInfoResult(packageInfoResult, (re) => {
          Logger.d("replyPackageInfoResult:", re)
        })

      } else if (result.event == NearbyEventType.transferNotify) {

        if (result.data.transferState == 6) {
          // todo文件接收完成，拷贝接收到的 patch文件到游戏脚本目录
        } else {
          Logger.d("传输状态：", result)
        }
      } else {
        Logger.d("状态监听：", result)
      }

    })

注销事件监听

  • API 定义
  /**
   * 注销附近传输服务的所有事件监听
   * @description 移除底层 `gameNearbyTransfer` 相关原生事件（如连接通知、传输通知、错误等）的监听函数，
   *              停止接收各类事件回调，避免内存泄漏或无效的事件处理
   * @note 该方法会注销所有已注册的原生事件监听，调用后 `registerCallback` 注册的全局回调将不再生效；
   *       若需重新接收事件，需再次调用 `registerCallback`
   * @throws 内部捕获事件注销过程中的异常并记录日志，不会向外抛出错误
   */
  public unregisterCallback()
  • 示例代码
NearbyTransferService.getInstance().unregisterCallback()
